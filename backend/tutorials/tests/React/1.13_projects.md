import { useState, useEffect } from "react";
import Count from "./components/count";


function App(){
	const [countA, setCountA] = useState(0);
	const [countB, setCountB] = useState(0);
	const [congra, setCongra] = useState(false);
	function resetAll(){
		
		setCountA(0),
		setCountB(0)
		
	}
	
	let total = countA + countB;

	useEffect(()=> {
		document.title = `Total: ${total}`;
	}, [countA, countB])
	
	useEffect(() => {
		console.log("Counter A changed")
	}, [countA])

	useEffect(() => {
		console.log("Counter B changed")
	}, [countB])


	useEffect(() => {
		if (total >= 20){
			setCongra(true)
		}else{
			setCongra(false)
		}
	}, [total])
	return(
		<div className="min-h-screen flex flex-col justify-center items-center  gap-6">
			<p className="text-5xl font-mono text-blue-500 font-bold text-center">Total: {total}</p>
			<div className="flex gap-4">
				<Count title="Counter A" counter="A" count={countA} setCount={setCountA}/>
				<Count title="Counter B" counter="B" count={countB} setCount={setCountB}/>
				
			</div>
			<button 
					className="bg-blue-500 px-4 py-2 text-white font-bold text-center font-mono hover:bg-blue-600"
					onClick={resetAll}
			>
				Reset All 
			</button>
			<p>{congra ? "Congratulation" : "" }</p>
		</div>
	)
}

export default App;



function Button({onClick, children}){
    return(
        <button 
            onClick={onClick}
            className="rounded-md px-4 py-2 bg-blue-500 text-white font-bold hover:bg-blue-600 transition-colors duration-300"
        >
            {children}
        </button>
    )
}

export default Button;



import Button from "./Button";

function Count({title, counter, count, setCount}){
    return (
        <div className="flex flex-col border shadow-lg rounded-xl px-4 py-2">
            <h2 className="text-4xl text-center font-bold">{title}</h2>
            <p className="text-2xl text-center font-bold">{count}</p>
            <div className="flex gap-4">
                <Button onClick= {() => setCount(prev => prev + 1)}>
                    Increase {counter}
                </Button>
                <Button onClick= {() => setCount(prev => prev - 1)}>
                    Decrease {counter}
                </Button>
                <Button onClick= {() => setCount(0)}>
                    Reset {counter}
                </Button>
            </div>
        </div>
    )
}

export default Count;
