import { useState } from "react";
import TaskCard from "./components/TaskCard";
import TaskList from "./components/TaskList";

function App() {
    const [tasks, setTasks] = useState([
        { id: 1, title: "Fix login bug", priority: "High", isCompleted: false },
        { id: 2, title: "Write tests", priority: "Medium", isCompleted: false },
        { id: 3, title: "Update docs", priority: "Low", isCompleted: false },
        { id: 4, title: "Deploy app", priority: "High", isCompleted: true },
        { id: 5, title: "Design mockup", priority: "Medium", isCompleted: true }
    ]);

    const [showHighPriorityOnly, setShowHighPriorityOnly] = useState(false);

    // Handler to toggle task completion status
    const handleToggleTask = (id) => {
        setTasks(prevTasks => prevTasks.map(task => 
            task.id === id ? { ...task, isCompleted: !task.isCompleted } : task
        ));
    };

    // Step 1: Filter tasks by priority if the toggle is active
    const prioritizedTasks = showHighPriorityOnly 
        ? tasks.filter(task => task.priority === "High") 
        : tasks;

    // Step 2: Split the prioritized tasks into active and completed lists
    const activeTasks = prioritizedTasks.filter(task => !task.isCompleted);
    const completedTasks = prioritizedTasks.filter(task => task.isCompleted);

    return (
        <div className="app-container">
            <h1>Task Management</h1>

            <button onClick={() => setShowHighPriorityOnly(prev => !prev)}>
                {showHighPriorityOnly ? "Show All Tasks" : "Show High Priority Tasks"}
            </button>

            <TaskList heading="Active Tasks">
                {activeTasks.map(task => (
                    <TaskCard 
                        key={task.id}
                        title={task.title}
                        priority={task.priority}
                        isCompleted={task.isCompleted}
                        onToggle={() => handleToggleTask(task.id)}
                    />
                ))}
            </TaskList>

            <TaskList heading="Completed Tasks">
                {completedTasks.map(task => (
                    <TaskCard 
                        key={task.id}
                        title={task.title}
                        priority={task.priority}
                        isCompleted={task.isCompleted}
                        onToggle={() => handleToggleTask(task.id)}
                    />
                ))}
            </TaskList>
        </div>
    );
}

export default App;
