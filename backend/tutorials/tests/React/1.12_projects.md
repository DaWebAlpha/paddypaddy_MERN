# PROJECT 7 - CONDITIONAL RENDERING

Type:

```jsx
import { useState } from "react";

function App() {

    const [loggedIn, setLoggedIn] = useState(false);

    return (
        <div>
            <h1>
                {loggedIn ? "Dashboard" : "Login"}
            </h1>

            <button
                onClick={() => setLoggedIn(!loggedIn)}
            >
                Toggle
            </button>
        </div>
    );
}

export default App;
```

import Button from "./Button"
function Count({countTitle, count, setCount}){
    return (
        <>
            <div className="flex flex-col shadow-md border border-gray-200 p-4 rounded-md">
                <h2 className="text-4xl font-mono font-bold text-center">count: {countTitle}</h2>
                <p className="text-2xl text-center font-mono font-bold">{count}</p>
                <div className="flex gap-4">
                    <Button onClick={() => setCount(prev => prev + 1)}>
                        Increase {countTitle}
                    </Button>

                    <Button onClick={() => setCount(prev => prev - 1)}>
                        Decrease {countTitle}
                    </Button>

                    <Button onClick={() => setCount(0)}>
                        Reset {countTitle}
                    </Button>
                </div>
            </div>
        </>
    )
}

export default Count
