## Mastery Assignment: Build a "Task Manager" Dashboard

Use **only** what we've learned: components, JSX, fragments, props, `&&`, ternary, and `useState`. No `useEffect`, no `.map()`, no events on custom components.

---

### Component 1: `TaskCard`

**Props:** `title` (string), `priority` (string: "High", "Medium", or "Low"), `isCompleted` (boolean)

**Render:**
- Title in an `<h3>`
- Priority with colored dot: red for "High", orange for "Medium", green for "Low" — use `&&` for each
- Priority text after the dot (just render `{priority}` directly)
- Status: "✓ Done" if completed, "○ Pending" if not — use ternary
- A `<button>` showing "Undo" if completed, "Mark Complete" if not — use ternary. **This button does nothing yet.**

**Default:** `isCompleted` defaults to `false` if not passed.

---

### Component 2: `TaskList`

**Props:** `heading` (string), `children`

**Render:**
- The `heading` in an `<h2>`
- A `<div>` that wraps `{children}`
- Use a fragment `<>...</>` as the root

---

### Component 3: `App`

**State:** Use `useState` to track `showHighPriorityOnly` (boolean, starts as `false`)

**Render:**
- A `<button>` that toggles `showHighPriorityOnly`. Label: `"Show High Only"` when `false`, `"Show All"` when `true`.
- Two `TaskList` instances:
  - **First:** heading `"Active Tasks"`, contains **3** `TaskCard` components
  - **Second:** heading `"Completed Tasks"`, contains **2** `TaskCard` components
- **Conditional display:** When `showHighPriorityOnly` is `true`, only show cards where `priority === "High"` in **both** lists. Use `&&` to conditionally render each `TaskCard`.

**Data requirements for the 5 cards:**
- All different `title`, `priority`, and `isCompleted` values
- Mix of priorities: at least one "High", one "Medium", one "Low" across both lists
- Mix of completed statuses

TaskCard.jsx
function TaskCard({ title, priority, isCompleted, onToggle }) {
  return (
    <div className="p-4 my-2 border border-gray-500 rounded-lg shadow-sm bg-white">
      <h3 className="text-xl font-semibold text-gray-800">{title}</h3>
      <p className="text-gray-600 mt-1">
        Priority: 
        {priority === "High" && <span style={{color:"red"}}>●</span>}
        {priority === "Medium" && <span style={{color:"orange"}}>●</span>}
        {priority === "Low" && <span style={{color:"green"}}>●</span>}
        {" "}{priority}
      </p>
      <p className="text-sm font-medium mt-1">{isCompleted ? "✓ Done" : "○ Pending"}</p>
      <button onClick={onToggle} className="mt-3 rounded-md bg-blue-500 text-white px-4 py-2 hover:bg-blue-600 font-medium text-sm">
        {isCompleted ? "Undo" : "Mark Complete"}
      </button>
    </div>
  );
}

export default TaskCard;

App.jsx
import { useState } from "react";
import TaskList from "./components/TaskList";
import TaskCard from "./components/TaskCard";

function App() {
  const [showHighPriorityOnly, setShowHighPriorityOnly] = useState(false);
  
  // One state per task
  const [isBugFixed, setIsBugFixed] = useState(false);
  const [isTestsWritten, setIsTestsWritten] = useState(false);
  const [isDocsUpdated, setIsDocsUpdated] = useState(false);
  const [isAppDeployed, setIsAppDeployed] = useState(true);
  const [isMockupDone, setIsMockupDone] = useState(true);

  return (
    <div>
      <h1>Task Management</h1>
      
      <button onClick={() => setShowHighPriorityOnly(prev => !prev)}>
        {showHighPriorityOnly ? "Show All" : "Show High Only"}
      </button>

      <TaskList heading="Active Tasks">
        {(!showHighPriorityOnly || "High" === "High") && (
          <TaskCard 
            title="Fix login bug" 
            priority="High" 
            isCompleted={isBugFixed} 
            onToggle={() => setIsBugFixed(prev => !prev)} 
          />
        )}
        {(!showHighPriorityOnly || "Medium" === "High") && (
          <TaskCard 
            title="Write tests" 
            priority="Medium" 
            isCompleted={isTestsWritten} 
            onToggle={() => setIsTestsWritten(prev => !prev)} 
          />
        )}
        {(!showHighPriorityOnly || "Low" === "High") && (
          <TaskCard 
            title="Update docs" 
            priority="Low" 
            isCompleted={isDocsUpdated} 
            onToggle={() => setIsDocsUpdated(prev => !prev)} 
          />
        )}
      </TaskList>

      <TaskList heading="Completed Tasks">
        {(!showHighPriorityOnly || "High" === "High") && (
          <TaskCard 
            title="Deploy app" 
            priority="High" 
            isCompleted={isAppDeployed} 
            onToggle={() => setIsAppDeployed(prev => !prev)} 
          />
        )}
        {(!showHighPriorityOnly || "Medium" === "High") && (
          <TaskCard 
            title="Design mockup" 
            priority="Medium" 
            isCompleted={isMockupDone} 
            onToggle={() => setIsMockupDone(prev => !prev)} 
          />
        )}
      </TaskList>
    </div>
  );
}

export default App;
